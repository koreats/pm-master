## 1. 디자인 원칙

### 1.1 핵심 가치

```typescript
interface DesignPrinciples {
  simplicity: '복잡함을 제거하고 본질에 집중'
  consistency: '일관된 경험으로 학습 곡선 최소화'
```

### 1.2 디자인 철학

- **Less is More**: 최소한의 요소로 최대한의 가치
- **User First**: 사용자 중심의 의사결정
- **Progressive Disclosure**: 단계적 정보 공개
- **Mobile First**: 모바일 우선 반응형 디자인

### 12.3 Push 알림 설정

```typescript
// Push 알림 권한 요청
const requestNotificationPermission = async () => {
  if (!('Notification' in window)) {
    console.log('This browser does not support notifications')
    return false
  }

  if (Notification.permission === 'granted') {
    return true
  }

  if (Notification.permission !== 'denied') {
    const permission = await Notification.requestPermission()
    return permission === 'granted'
  }

  return false
}

// Push 알림 표시
```

```typescript
function ThemeToggle() {
  const { theme, setTheme } = useTheme();
  const [isTransitioning, setIsTransitioning] = useState(false);

  const handleThemeChange = () => {
    setIsTransitioning(true);
    setTheme(theme === 'dark' ? 'light' : 'dark');

    // 애니메이션 완료 후 클래스 제거
    setTimeout(() => {
      setIsTransitioning(false);
    }, 200);
  };

  return (
    <button
      onClick={handleThemeChange}
      className={cn(
        'theme-toggle',
        isTransitioning && 'transitioning'
      )}
    >
      {theme === 'dark' ? '🌞' : '🌙'}
    </button>
  );
}
```

```tsx
// 접근성을 고려한 컴포넌트 예시
function AccessibleButton({ children, onClick, ...props }) {
  return (
    <button
      onClick={onClick}
      aria-label={props['aria-label']}
      aria-pressed={props['aria-pressed']}
      aria-disabled={props.disabled}
      role="button"
      tabIndex={props.disabled ? -1 : 0}
      onKeyDown={e => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault()
          onClick(e)
        }
      }}
      className={cn('focus:outline-none focus:ring-2 focus:ring-offset-2', props.className)}
      {...props}
    >
      {children}
    </button>
  )
}
```
